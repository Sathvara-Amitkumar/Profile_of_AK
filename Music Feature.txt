/*=============== MUSIC PLAYER (dynamic island) ===============*/
const TRACKS = [   // naye gaane yaha add karo
  { src: 'assets/audio/bairan.mp3', title: 'Bairan', artist: 'Banjaare', cover: 'assets/img/cover-bairan.jpg' },
  { src: 'assets/audio/tere-bina.mp3', title: 'Tere Bina', artist: 'A R Rahman', cover: 'assets/img/cover-tere-bina.jpg' }
];
(() => {
  const $ = id => document.getElementById(id), island = $('island'), audio = new Audio();
  audio.preload = 'none'; let n = 0;
  const fmt = s => isFinite(s) ? `${Math.floor(s / 60)}:${String(Math.floor(s % 60)).padStart(2, '0')}` : '0:00';
  const load = (i, play) => { n = (i + TRACKS.length) % TRACKS.length; const t = TRACKS[n];
    audio.src = t.src; $('p-title').textContent = t.title; $('p-artist').textContent = t.artist; $('p-cover').src = t.cover;
    $('p-seek').value = 0; $('p-cur').textContent = '0:00'; $('p-dur').textContent = '0:00'; if (play) audio.play(); };
  const setIcon = () => { const p = !audio.paused; island.classList.toggle('playing', p); $('p-play').innerHTML = `<i class="ri-${p ? 'pause' : 'play'}-fill"></i>`; };
  load(0, false);
  island.addEventListener('click', e => {
    if (!island.classList.contains('open')) island.classList.add('open');
    else if (e.target.closest('.island__top')) island.classList.remove('open');
  });
  document.addEventListener('click', e => { if (!e.target.closest('#island')) island.classList.remove('open'); });
  $('p-play').onclick = () => audio.paused ? audio.play() : audio.pause();
  $('p-next').onclick = () => load(n + 1, true);
  $('p-prev').onclick = () => load(n - 1, true);
  $('p-seek').oninput = e => audio.currentTime = (audio.duration * e.target.value / 100) || 0;
  audio.onplay = audio.onpause = setIcon;
  audio.onloadedmetadata = () => $('p-dur').textContent = fmt(audio.duration);
  audio.ontimeupdate = () => { $('p-cur').textContent = fmt(audio.currentTime); $('p-seek').value = audio.duration ? audio.currentTime / audio.duration * 100 : 0; };
  audio.onended = () => load(n + 1, true);
})();
